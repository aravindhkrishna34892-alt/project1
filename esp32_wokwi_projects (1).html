1. Temperature & Humidity Monitoring System (DHT22 + OLED)

Reads temperature and humidity from a DHT22 sensor and displays the readings on a 0.96" SSD1306 OLED screen.

Circuit Diagram

Wiring Table

ESP32 PinConnects To
3V3DHT22 VCC, OLED VCC
GNDDHT22 GND, OLED GND
GPIO15DHT22 OUT (data)
GPIO21OLED SDA
GPIO22OLED SCL
Required Wokwi/Arduino libraries: DHT sensor library, Adafruit Unified Sensor, Adafruit GFX Library, Adafruit SSD1306.

Code (sketch.ino)

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <DHT.h>

#define DHTPIN 15
#define DHTTYPE DHT22

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);
DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(115200);
  dht.begin();

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("SSD1306 allocation failed");
    while (true);
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println("Temp Monitor");
  display.display();
  delay(1000);
}

void loop() {
  float temp = dht.readTemperature();
  float hum = dht.readHumidity();

  if (isnan(temp) || isnan(hum)) {
    Serial.println("Failed to read from DHT sensor!");
    display.clearDisplay();
    display.setCursor(0, 0);
    display.setTextSize(1);
    display.println("Sensor Error!");
    display.display();
    delay(2000);
    return;
  }

  Serial.printf("Temp: %.1f C, Humidity: %.1f %%\n", temp, hum);

  display.clearDisplay();

  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Temperature Monitor");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  display.setTextSize(2);
  display.setCursor(0, 20);
  display.print("T: ");
  display.print(temp, 1);
  display.print(" C");

  display.setCursor(0, 45);
  display.print("H: ");
  display.print(hum, 1);
  display.print(" %");

  display.display();
  delay(2000);
}

2. Gas Leak Alert System (MQ-2 + OLED + Buzzer + LED)

Reads gas concentration from an MQ-2 sensor, shows the live value on an OLED, and triggers a buzzer + LED when the level exceeds a threshold.

Circuit Diagram

Wiring Table

ESP32 PinConnects To
5VMQ-2 VCC
GNDMQ-2 GND, Buzzer (-), LED (-), OLED GND
GPIO34MQ-2 AOUT
GPIO4Buzzer (+)
GPIO2LED (+) through resistor
GPIO21 (SDA)OLED SDA
GPIO22 (SCL)OLED SCL
3V3OLED VCC
Required libraries: Adafruit GFX Library, Adafruit SSD1306. The MQ-2 in Wokwi has a slider to simulate gas concentration — drag it up during simulation to trigger the alert.

Code (sketch.ino)

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define MQ2_PIN 34
#define BUZZER_PIN 4
#define LED_PIN 2

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

const int GAS_THRESHOLD = 1500; // tune based on testing

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("SSD1306 allocation failed");
    while (true);
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println("Gas Alert System");
  display.display();
  delay(1000);
}

void loop() {
  int gasValue = analogRead(MQ2_PIN);
  Serial.print("Gas level: ");
  Serial.println(gasValue);

  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Gas Monitor");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  display.setTextSize(2);
  display.setCursor(0, 20);
  display.print("Lvl: ");
  display.println(gasValue);

  display.setTextSize(1);
  display.setCursor(0, 50);

  if (gasValue > GAS_THRESHOLD) {
    display.println("ALERT: Gas detected!");
    digitalWrite(LED_PIN, HIGH);
    digitalWrite(BUZZER_PIN, HIGH);
  } else {
    display.println("Status: Normal");
    digitalWrite(LED_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
  }

  display.display();
  delay(500);
}

3. Ultrasonic Parking Sensor (HC-SR04 + Buzzer + LED + OLED)

Measures distance using an HC-SR04 ultrasonic sensor, shows it on an OLED, and warns with LED/buzzer as an object gets closer.

Circuit Diagram

Wiring Table

ESP32 PinConnects To
5VHC-SR04 VCC
GNDHC-SR04 GND, Buzzer (-), LED (-), OLED GND
GPIO5HC-SR04 Trig
GPIO18HC-SR04 Echo
GPIO4Buzzer (+)
GPIO2LED (+) through resistor
GPIO21 (SDA)OLED SDA
GPIO22 (SCL)OLED SCL
3V3OLED VCC
Required libraries: Adafruit GFX Library, Adafruit SSD1306. In Wokwi, drag the HC-SR04's distance slider during simulation to test near/far behavior.

Code (sketch.ino)

#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

#define TRIG_PIN 5
#define ECHO_PIN 18
#define BUZZER_PIN 4
#define LED_PIN 2

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  if (!display.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("SSD1306 allocation failed");
    while (true);
  }

  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println("Parking Sensor");
  display.display();
  delay(1000);
}

void loop() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);

  long duration = pulseIn(ECHO_PIN, HIGH);
  float distance = duration * 0.0343 / 2;

  Serial.print("Distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 0);
  display.println("Parking Assist");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);

  display.setTextSize(2);
  display.setCursor(0, 20);
  display.print(distance, 1);
  display.println(" cm");

  display.setTextSize(1);
  display.setCursor(0, 50);

  if (distance < 10) {
    display.println("STOP! Too close!");
    digitalWrite(LED_PIN, HIGH);
    digitalWrite(BUZZER_PIN, HIGH);
    delay(100);
    digitalWrite(BUZZER_PIN, LOW);
    delay(100);
  } else if (distance < 30) {
    display.println("Slow down");
    digitalWrite(LED_PIN, HIGH);
    digitalWrite(BUZZER_PIN, LOW);
  } else {
    display.println("Clear");
    digitalWrite(LED_PIN, LOW);
    digitalWrite(BUZZER_PIN, LOW);
  }

  display.display();
  delay(200);
}
